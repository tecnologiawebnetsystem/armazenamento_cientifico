3) BANCO DE DADOS (modelo sugerido)
------------------------------------------------------------------------------------
  perfil (id, nome, descricao)
  perfil_permissao (perfil_id -> perfil.id, permissao TEXT)
  usuario (... perfil_id -> perfil.id)

  Consulta única (efetiva):
    SELECT p.nome, array_agg(pp.permissao)
    FROM usuario u
    JOIN perfil p ON p.id = u.perfil_id
    LEFT JOIN perfil_permissao pp ON pp.perfil_id = p.id
    WHERE u.id = :user_id
    GROUP BY p.nome;
